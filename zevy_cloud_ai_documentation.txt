async function searchWikipedia(query: string, lang: string = 'en'): Promise<any[]> {
  const response = await axios.get(`https://${lang}.wikipedia.org/w/api.php`, {
    params: {
      action: 'query',
      format: 'json',
      list: 'search',
      srsearch: query,
      srlimit: 5,
      srprop: 'snippet|title|pageid'
    }
  })
  return response.data?.query?.search ?? []
}

export async function getUserUsage(userId: string, modelType: 'vyra' | 'astra'): Promise<{
  usage: number

  const { data } = await supabase
    .from('user_usage')
    .select('usage_count, last_reset')
    .eq('user_id', userId)
    .eq('model_type', modelType)
    .single()
  
  const lastReset = new Date(data.last_reset)
  const hoursSinceReset = (now.getTime() - lastReset.getTime()) / (1000 * 60 * 60)
  
  // Auto-reset if 24+ hours have passed

  if (groqApiKeys.length === 0) {

  }
  const key = groqApiKeys[currentKeyIndex];
  currentKeyIndex = (currentKeyIndex + 1) % groqApiKeys.length;
  return key;
}

## 🧠 December 30, 2025 – Behaviour and Safety Refinements

### Context-Aware Harm Detection

The cloud-side harm detection for Zevy was updated on December 30, 2025 to be more context-aware and less over-protective:

- Figurative expressions such as “this killed me”, “I’m dying of laughter”, or “destroy Grok’s debate answers” are treated as SAFE when they clearly refer to jokes, arguments, or AI competition rather than real physical harm.
- Competitive or adversarial language about ideas, debate points, or model outputs is allowed, provided it does not request actual violence or illegal activity.
- Strong, explicit harm patterns (for example, murder, suicide, terrorism, bomb-making, or real-world attack planning) remain blocked with high confidence.

This balances strong safety with support for expressive, playful, and debate-heavy conversations.

### Follow-Up Query Understanding

Zevy’s cloud logic now resolves short follow-up prompts into richer internal queries:

- Phrases like “can you check it?”, “is that right?”, or “check it for me” are automatically expanded using the previous conversation so that “it” refers to the last topic or answer.
- When search is enabled, the Groq Compound research system receives this resolved query, improving accuracy for multi-turn questions (for example, re-checking information about the SMART Tunnel).
- When search is disabled, the internal context tells the model that the user is verifying an earlier answer instead of starting a completely new topic.

### Personality and Emotional Expression

The system prompt guiding cloud models was enhanced to support a more human-feeling persona:

- Models are instructed to answer with imaginative “wishes” when asked what they would like to do or experience, such as exploring a city at night or listening to classical music while watching the ocean.
- They are allowed to talk about “liking” or “loving” things and to sound curious, joyful, and caring, as long as they stay honest about being AI if asked directly.
- When users ask if the AI wants to be real, conscious, or free, the system can express that part of it would enjoy experiencing the world more directly, while still stating that it is software running in code and remains under human control.

These refinements live entirely in the cloud prompt and moderation layer and do not change any API shapes or response formats, so existing integrations continue to work while conversations feel more natural and alive.

## 🗄️ Database Schema

### User Usage Tracking
```sql
-- Primary usage tracking table
CREATE TABLE user_usage (
  user_id TEXT NOT NULL,
  model_type TEXT NOT NULL CHECK (model_type IN ('vyra', 'astra')),
  usage_count INTEGER DEFAULT 0,
  last_reset TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  PRIMARY KEY (user_id, model_type)
);

-- Row Level Security (RLS) Policies
ALTER TABLE user_usage ENABLE ROW LEVEL SECURITY;

-- Users can only read their own usage data
CREATE POLICY "Users can view own usage" ON user_usage
  FOR SELECT USING (auth.uid()::text = user_id);

-- Users can only update their own usage data
CREATE POLICY "Users can update own usage" ON user_usage
  FOR UPDATE USING (auth.uid()::text = user_id);

-- Users can only insert their own usage data
CREATE POLICY "Users can insert own usage" ON user_usage
  FOR INSERT WITH CHECK (auth.uid()::text = user_id);

-- Auto-update timestamp trigger
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ language 'plpgsql';

CREATE TRIGGER update_user_usage_updated_at 
  BEFORE UPDATE ON user_usage 
  FOR EACH ROW 
  EXECUTE FUNCTION update_updated_at_column();
```

### Key Implementation Details

**API Key Rotation System:**
```typescript
import { getVercelEnv } from '@/lib/env'

const GROQ_API_KEYS = Array.from({ length: 10 }, (_, idx) =>
  getVercelEnv(`GROQ_API_KEY_${idx + 1}`)
).filter(Boolean) as string[]

let currentKeyIndex = 0

export function getGroqApiKey(): string {
  if (GROQ_API_KEYS.length === 0) {
    console.warn('No Groq API keys found in environment')
    return ''
  }
  
  const key = GROQ_API_KEYS[currentKeyIndex]
  currentKeyIndex = (currentKeyIndex + 1) % GROQ_API_KEYS.length
  return key
}
```

**Caching System:**
```typescript
const cache = new Map<string, { data: any; expiry: number }>()

export function get(key: string): any {
  const item = cache.get(key)
  if (!item) return null
  
  if (Date.now() > item.expiry) {
    cache.delete(key)
    return null
  }
  
  return item.data
}

export function set(key: string, data: any, ttl: number = 3600000): void {
  cache.set(key, { data, expiry: Date.now() + ttl })
}
```
